Authentication Modal Logic & Tabs Switch const authModal = document.getElementById('auth-modal'); const authModalCard = document.getElementById('auth-modal-card'); const loginForm = document.getElementById('login-form'); const registerForm = document.getElementById('register-form'); const tabLogin = document.getElementById('tab-login'); const tabRegister = document.getElementById('tab-register'); function openAuthModal(tab = 'login') { switchAuthTab(tab); authModal.classList.remove('hidden'); setTimeout(() => { authModal.classList.remove('opacity-0'); authModalCard.classList.remove('scale-95'); }, 10); } function closeAuthModal() { authModal.classList.add('opacity-0'); authModalCard.classList.add('scale-95'); setTimeout(() => { authModal.classList.add('hidden'); }, 300); } function switchAuthTab(tab) { if (tab === 'login') { loginForm.classList.remove('hidden'); registerForm.classList.add('hidden'); tabLogin.className = 'flex-1 py-3 font-bold text-center border-b-2 border-brand-600 text-brand-600 transition-colors'; tabRegister.className = 'flex-1 py-3 font-semibold text-center border-b-2 border-transparent text-slate-400 hover:text-slate-600 transition-colors'; } else { loginForm.classList.add('hidden'); registerForm.classList.remove('hidden'); tabRegister.className = 'flex-1 py-3 font-bold text-center border-b-2 border-brand-600 text-brand-600 transition-colors'; tabLogin.className = 'flex-1 py-3 font-semibold text-center border-b-2 border-transparent text-slate-400 hover:text-slate-600 transition-colors'; } } // Close Modal when clicking backdrop authModal.addEventListener('click', (e) => { if (e.target === authModal) { closeAuthModal(); } }); // 7. Simulated User Login / Sign Up State function handleAuthSubmit(event, type) { event.preventDefault(); let userName = "Alex Johnson"; if (type === 'register') { userName = document.getElementById('reg-name').value || "Alex Johnson"; } else { const email = document.getElementById('login-email').value; userName = email.split('@')[0] || "User"; userName = userName.charAt(0).toUpperCase() + userName.slice(1); } // Get initials const initials = userName.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2) || "U"; // Update UI to logged-in state document.getElementById('user-display-name').textContent = userName; document.getElementById('user-avatar-initials').textContent = initials; document.getElementById('auth-container').classList.add('hidden'); document.getElementById('user-logged-in-container').classList.remove('hidden'); document.getElementById('user-logged-in-container').classList.add('flex'); closeAuthModal(); showToast(`Welcome back, ${userName}! Logged in successfully.`); } function logoutUser() { document.getElementById('auth-container').classList.remove('hidden'); document.getElementById('user-logged-in-container').classList.add('hidden'); document.getElementById('user-logged-in-container').classList.remove('flex'); showToast("You have been logged out."); }
